  <script>
    let students = JSON.parse(localStorage.getItem('5c_students')) || Array.from({length: 30}, (_, i) => ({
      id: i,
      name: `Student ${i+1}`,
      score: 0
    }));

    let soundEnabled = true;
    let historyStack = [];
    let editingStudentId = null;
    let dragSrcIndex = null;

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    function playTone(freq, type, duration) {
      if (!soundEnabled) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch(e){}
    }

    function playFanfare() {
      if (!soundEnabled) return;
      const notes = [261.63, 329.63, 392.00, 523.25];
      notes.forEach((freq, idx) => {
        setTimeout(() => playTone(freq, 'triangle', 0.3), idx * 120);
      });
    }

    function toggleSound() {
      soundEnabled = !soundEnabled;
      document.getElementById('soundBtn').innerText = soundEnabled ? "🔊 Sound ON" : "🔇 Sound OFF";
    }

    function saveState() {
      localStorage.setItem('5c_students', JSON.stringify(students));
    }

    function renderBoard() {
      const grid = document.getElementById('seatingGrid');
      grid.innerHTML = '';

      for (let r = 0; r < 6; r++) {
        const rowCol = document.createElement('div');
        rowCol.className = 'row-column';

        const rowStudents = students.slice(r * 5, (r + 1) * 5);
        const rowTotal = rowStudents.reduce((sum, s) => sum + s.score, 0);

        rowCol.innerHTML = `
          <div class="row-header">
            <span class="row-title">Row ${r + 1}</span>
            <span class="row-score">Total: ${rowTotal} pts</span>
            <button class="btn-row-add" onclick="addPointToRow(${r})">+1 All in Row</button>
          </div>
        `;

        rowStudents.forEach((st, idx) => {
          const globalIdx = r * 5 + idx;
          const card = document.createElement('div');
          card.className = 'student-card';
          card.draggable = true;
          card.dataset.index = globalIdx;

          card.innerHTML = `
            <div class="student-name" onclick="openEditModal(${st.id})">${st.name}</div>
            <div class="student-score">${st.score}</div>
            <div class="card-actions">
              <button class="btn-pt btn-sub" onclick="changeScore(${st.id}, -1)">-1</button>
              <button class="btn-pt btn-add" onclick="changeScore(${st.id}, 1)">+1</button>
            </div>
          `;

          // Drag and drop
          card.addEventListener('dragstart', (e) => { dragSrcIndex = globalIdx; });
          card.addEventListener('dragover', (e) => { e.preventDefault(); card.classList.add('drag-over'); });
          card.addEventListener('dragleave', () => { card.classList.remove('drag-over'); });
          card.addEventListener('drop', (e) => {
            e.preventDefault();
            card.classList.remove('drag-over');
            if (dragSrcIndex !== null && dragSrcIndex !== globalIdx) {
              const temp = students[dragSrcIndex];
              students[dragSrcIndex] = students[globalIdx];
              students[globalIdx] = temp;
              saveState();
              renderBoard();
            }
          });

          rowCol.appendChild(card);
        });

        grid.appendChild(rowCol);
      }
    }

    function changeScore(id, delta) {
      const st = students.find(s => s.id === id);
      if (!st) return;

      historyStack.push(JSON.stringify(students));
      if (historyStack.length > 20) historyStack.shift();

      const oldScore = st.score;
      st.score = Math.max(0, st.score + delta);

      if (delta > 0 && st.score > 0 && st.score % 5 === 0 && st.score !== oldScore) {
        triggerCelebration(st.name, st.score);
      } else if (delta > 0) {
        playTone(523.25, 'sine', 0.15);
      }

      saveState();
      renderBoard();
    }

    function addPointToRow(rowIndex) {
      historyStack.push(JSON.stringify(students));
      for (let i = rowIndex * 5; i < (rowIndex + 1) * 5; i++) {
        students[i].score += 1;
        if (students[i].score % 5 === 0) {
          triggerCelebration(students[i].name, students[i].score);
        }
      }
      playFanfare();
      saveState();
      renderBoard();
    }

    function undoLastAction() {
      if (historyStack.length === 0) return;
      students = JSON.parse(historyStack.pop());
      saveState();
      renderBoard();
    }

    function triggerCelebration(name, score) {
      playFanfare();
      confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });

      const titles = ["🌟 SUPER STAR!", "👏 FANTASTIC!", "🎉 AWESOME JOB!", "🔥 BRILLIANT!"];
      const randomTitle = titles[Math.floor(Math.random() * titles.length)];

      document.getElementById('celebTitle').innerText = randomTitle;
      document.getElementById('celebMsg').innerText = `${name} earned ${score} Stars! 🎀🐟`;
      document.getElementById('celebrateModal').classList.add('active');
    }

    function openRosterModal() {
      document.getElementById('rosterInput').value = students.map(s => s.name).join('\n');
      document.getElementById('rosterModal').classList.add('active');
    }

    function saveRoster() {
      const text = document.getElementById('rosterInput').value;
      const names = text.split(/[\n,]/).map(n => n.trim()).filter(n => n.length > 0);
      
      names.forEach((name, i) => {
        if (i < 30) students[i].name = name;
      });

      saveState();
      renderBoard();
      closeModal('rosterModal');
    }

    function openEditModal(id) {
      editingStudentId = id;
      const st = students.find(s => s.id === id);
      document.getElementById('editNameInput').value = st.name;
      document.getElementById('editModal').classList.add('active');
    }

    function confirmEditName() {
      const newName = document.getElementById('editNameInput').value.trim();
      if (newName && editingStudentId !== null) {
        const st = students.find(s => s.id === editingStudentId);
        if (st) st.name = newName;
        saveState();
        renderBoard();
      }
      closeModal('editModal');
    }

    function pickRandomStudent() {
      let count = 0;
      const interval = setInterval(() => {
        const randomIdx = Math.floor(Math.random() * 30);
        const name = students[randomIdx].name;
        document.getElementById('celebTitle').innerText = "🎲 RANDOM PICKER";
        document.getElementById('celebMsg').innerText = `Selecting... ${name}`;
        document.getElementById('celebrateModal').classList.add('active');
        playTone(400 + count * 20, 'sine', 0.05);
        count++;
        if (count > 15) {
          clearInterval(interval);
          playFanfare();
          document.getElementById('celebMsg').innerText = `👉 ${name} 👈`;
        }
      }, 100);
    }

    function startTimerPrompt() {
      const sec = prompt("Enter seconds for timer (e.g., 30, 60, 120):", "60");
      if (!sec || isNaN(sec)) return;
      let time = parseInt(sec);
      
      document.getElementById('celebTitle').innerText = "⏱️ TIMER";
      document.getElementById('celebrateModal').classList.add('active');
      
      const timerInt = setInterval(() => {
        document.getElementById('celebMsg').innerText = `Time Remaining: ${time}s`;
        if (time <= 0) {
          clearInterval(timerInt);
          playFanfare();
          document.getElementById('celebMsg').innerText = "⏰ TIME'S UP!";
        }
        time--;
      }, 1000);
    }

    function openLeaderboard() {
      const sortedStudents = [...students].sort((a,b) => b.score - a.score).slice(0, 5);
      
      let rowScores = [];
      for(let r=0; r<6; r++) {
        const sum = students.slice(r*5, (r+1)*5).reduce((a,b) => a + b.score, 0);
        rowScores.push({ row: r+1, score: sum });
      }
      rowScores.sort((a,b) => b.score - a.score);

      let html = `<strong>🥇 Top Row:</strong> Row ${rowScores[0].row} (${rowScores[0].score} pts)<br><br>`;
      html += `<strong>🏆 Top 5 Students:</strong><ol style="margin-left:20px;">`;
      sortedStudents.forEach(s => {
        html += `<li>${s.name}: ${s.score} stars</li>`;
      });
      html += `</ol>`;

      document.getElementById('leaderboardContent').innerHTML = html;
      document.getElementById('leaderboardModal').classList.add('active');
    }

    function resetNewWeek() {
      if (confirm("Reset all scores for a new week? Student names will be kept.")) {
        students.forEach(s => s.score = 0);
        saveState();
        renderBoard();
        closeModal('leaderboardModal');
      }
    }

    function closeModal(id) {
      document.getElementById(id).classList.remove('active');
    }

    renderBoard();
  </script>
